  for(int i=1; i<=n; i++){
    int a; scanf("%d", &a);
    if(turn == 1)
      ans[end++] = a;
    else
      ans[--start] = a;
    turn *= (-1);

    for(int i=end-1; i>=start; i--)
      printf("%d ", ans[i]);
  else
    for(int i=start; i<end; i++)
      printf("%d ", ans[i]);
  puts("");

  return 0;
}
