void SwapNum_Void(U32 *u32_x, U32 *u32_y)
{
									/* 一時保持変数 */
	U32 u32_tmp = 0;				/* 変数定義 */
	
	u32_tmp = *u32_x;				/* u32_xの値を保持 */
	*u32_x  = *u32_y;				/* u32_yの値をu32_xに格納 */
	*u32_y  = u32_tmp;				/* 保持していたu32_xの値をu32_yに格納 */
}

	<引数>

	<補足説明>

U32 CalcGcd_U32(U32 u32_x, U32 u32_y)
{
									/* x%yの結果を保持する変数 */
	U32 u32_rem = 0;				/* 変数定義 */
	
	/* ヒントによると、「xとyの最大公約数は、yとx%yの最大公約数と一致する」らしい */
	/* → x%yの結果がゼロになるまでxにyの値を、yにx%yの結果を格納し続け、 */
	/*    x%yの結果がゼロになった時のyが最大公約数 */
	while(1)						/* x%yの結果がゼロではない間ループ */
	{
		u32_rem = u32_x % u32_y;	/* x%yを計算 */
		
		if(0 == u32_rem)			/* x%yの結果がゼロの場合 */
		{
			break;
		}
		
		u32_x = u32_y;				/* u32_xを更新 */
		u32_y = u32_rem;			/* u32_yを更新 */
	}
	
	return u32_y;
}
