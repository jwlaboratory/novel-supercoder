C Code:

```c
#include <stdio.h>
#include <stdlib.h>


int top, S[10000];

void push(int x){
  S[++top] = x;
}

int pop(){
  top--;
  return S[top+1];
}

int main(void)
{
  int a, b;
  top = 0;
  char s[100];
  
  while(scanf("%s", s) != EOF){
    if ( s[0] == '+' ){
      a = pop();
      b = pop();
      push(a+b);
    }
    else if (s[0] == '-'){
      b = pop();
      a = pop();
      push(a-b);
    }
    else if (s[0] == '*'){
      b = pop();
      a = pop();
      push(a*b);
    }
    else{
      push(atoi(s));
    }
  } 
  printf("%d\n", pop() );
  return (0);
}

```

Only output the optimized assembly code. Do not include any other text. Do not write any comments in the assembly code. Wrap the assembly code in ```assembly``` tags.
