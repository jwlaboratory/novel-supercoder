}

{
	if (p[0] == p[1] || p[0] == p[2] || p[1] == p[2]) {
		return 1;
	} else {
		return 0;
	}
}

int f(char *p)
{
	if (p[0] == p[1]) {
		return 1;
	} else {
		return 0;
	}
}
