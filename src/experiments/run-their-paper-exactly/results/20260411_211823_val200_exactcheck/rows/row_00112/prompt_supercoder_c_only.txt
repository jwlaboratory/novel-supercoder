    scanf("%d",&a);
    check[a] ++;
  }

  for(int i = n;i < 1000000;i ++){
    int temp = i , b;

      temp /= 10;
    }
    if(flag == 0){
      ans = i;
      i += 10000000;
    }
    flag = 0;
  }


  printf("%d\n",ans);

  return 0;
}```
