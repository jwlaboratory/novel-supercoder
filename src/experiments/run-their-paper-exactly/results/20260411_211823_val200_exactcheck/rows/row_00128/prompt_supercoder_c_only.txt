#include <math.h>
#include <time.h>

#include<string.h>

    scanf("%d %d %d %d %d", &a[0], &a[1], &a[2], &a[3], &a[4]);

    for (i = 0; i < 5; i++) {

        }

        }
        ans += (a[i] / 10) * 10 + 10;
    }
    ans -= loss;
    printf("%d", ans);
    return 0;
}```
