```assembly
section .text
global main
extern printf, scanf

main:
    push rbp
    mov rbp, rsp
    sub rsp, 32

    lea rdi, [LC0]
    lea rsi, [rbp-16]
    call scanf

    lea rdi, [LC1]
    lea rsi, [rbp-16]
    call scanf

    movzx eax, byte [rbp-16+rsi*1]
    cmp al, 'A'
    jb .lowercase
    cmp al, 'Z'
    ja .lowercase
    add al, 32
    jmp .print
.lowercase:
    sub al, 32
.print:
    movzx esi, al
    lea rdi, [rbp-16]
    xor eax, eax
    call printf

    mov rsp, rbp
    pop rbp
    ret
```
