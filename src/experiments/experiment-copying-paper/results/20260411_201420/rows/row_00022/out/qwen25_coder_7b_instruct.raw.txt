```assembly
section .text
global main
extern printf, scanf

main:
    push rbp
    mov rbp, rsp
    sub rsp, 128
    lea rdi, [rbp-12]
    xor eax, eax
    call scanf
    jnz .loop

    mov ecx, 30
    xor ebx, ebx
    lea rsi, [rbp-12]

.loop:
    cmp ecx, 0
    je .end_loop
    dec ecx
    mov eax, [rsi+4*ecx]
    test eax, eax
    jz .continue
    inc ebx
    xor eax, eax
    add eax, ecx
    shl eax, cl
    xor eax, [rbp-12]
.continue:
    jmp .loop

.end_loop:
    mov eax, ebx
    test eax, eax
    jz .print_result
    mov eax, -1
.print_result:
    mov edi, eax
    xor eax, eax
    call printf
    leave
    ret
```
