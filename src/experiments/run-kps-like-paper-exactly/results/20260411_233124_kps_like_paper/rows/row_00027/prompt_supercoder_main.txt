typedef struct tagItem Item;
struct tagItem {
    int value;

};

void push(Item **sp, int value)
{
    //printf("push %d\n", value);
    Item *item = (Item *)malloc(sizeof(Item));
    item->value = value;
    item->prev = *sp;
    *sp = item;
}

int pop(Item **sp)
{
    Item *item = *sp;
    if (item != NULL) {
        int value = item->value;
        *sp = item->prev;
        free(item);
        //printf("pop %d\n", value);
        return value;
    }
    //printf("pop error\n");
    return 0;
}
